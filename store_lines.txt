 556:async function getResolvedEnvironmentVariables(
 557:  environment: { name: string; variables: VariableDefinition[] },
 558:  getState: () => AppState,
 559:  setState: (partial: Partial<AppState>) => void,
 560:) {
 561:  const state = getState();
 562:  const resolved: VariableDefinition[] = [];
 563:  for (const variable of environment.variables) {
 564:    if (!variable.secret) {
 565:      resolved.push(variable);
 566:      continue;
 567:    }
 568:    const key = secretKey(environment.name, variable.name);
 569:    const cache = state.secretCache[key];
 570:    let value = cache?.value;
 571:    if (!value) {
 572:      const master = state.masterPassword;
 573:      if (!master) {
 574:        throw new Error("Master password requerida para descifrar secrets");
 575:      }
 576:      const encrypted = cache?.encrypted ?? variable.value;
 577:      value = await decryptSecret(master, encrypted);
 578:      setState((prev) => ({
 579:        secretCache: {
 580:          ...prev.secretCache,
 581:          [key]: {
 582:            environment: environment.name,
 583:            variable: variable.name,
 584:            encrypted,
 585:            value,
 586:          },
 587:        },
 588:      }));
 589:    }
 590:    resolved.push({ ...variable, value });
 591:  }
 592:  return resolved;
 593:}
