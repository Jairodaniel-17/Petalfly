   1:import { useEffect, useState } from "react";

   4:
   5:const SECRET_PLACEHOLDER = "••••••";
   6:
   7:export function SecretManager() {

  12:    (s) => s.activeEnvironment ?? s.settings.activeEnvironment ?? s.environments[0]?.name,

  68:        return;
  69:      }

  72:      ...prev,
  73:      [variable.name]: !prev[variable.name],
  74:    }));
  75:  };
  76:

  79:    setSecretValue(env.name, variable.name, value);

 101:      setMessages((error as Error).message);
 102:    }

 165:              <div className="secret-row" key={secret.name}>
 166:                <div className="secret-row__info">
 167:                  <strong>{secret.name}</strong>

 169:                    <input

 184:                    {visible ? "🙈" : "👁"}

 187:                    className="icon-button"
 188:                    title="Vaciar valor"
 189:                    onClick={() => handleClear(secret)}

 195:                    title="Eliminar secret"

 199:                  </button>
 200:                </div>
 201:              </div>

 206:          <input
 207:            placeholder="Nombre del secret"
 208:            value={newSecretName}
 209:            onChange={(event) => setNewSecretName(event.target.value)}
 210:          />
 211:          <input
 212:            placeholder="Valor"
 213:            value={newSecretValue}
 214:            onChange={(event) => setNewSecretValue(event.target.value)}
 215:          />
 216:          <button onClick={handleAddSecret}>Añadir secret</button>
 217:        </div>

 220:        </div>
 221:      </div>
 222:    </div>
 223:  );
